{{- /* Keyword coloring (see data/keywords.yaml). Called with (dict "page" . "content" .Content). Each page picks its own terms in front matter: colorKeywords = ["fuzzing", "supply chain"] # looked up in the library colorKeywordsCustom = { "AFL++" = "flood" } # page-only term = style Returns the content with the first mention of each picked term wrapped in , and warns at build time about picked terms that never got colored. Regex over HTML is only safe if we never touch markup, so the HTML is split into tokens: tags, comments, and the plain text between them. Text is only edited when it isn't inside a protected element (code, links, headings, ...), and tags themselves (including attributes like href and alt) are never edited. Go's regex engine has no lookbehind, so this token walk is what keeps matches out of protected regions. Matches are first swapped for placeholders (U+E000 index U+E001, private-use characters that never appear in content) and only turned into spans at the end, so a later keyword can't match text inside an earlier keyword's span. Keywords are tried longest first. */ -}} {{- $page := .page -}} {{- $html := string .content -}} {{- $data := hugo.Data.keywords -}} {{- /* Never colored inside these. Tables are data (and a summary table near the top would otherwise use up the first mention of every term before the prose gets to it). */ -}} {{- $protected := slice "a" "code" "pre" "kbd" "samp" "h1" "h2" "h3" "h4" "h5" "h6" "table" "script" "style" "svg" "button" "textarea" "select" "mark" -}} {{- /* Collect this page's picks. Library terms are found by any of their forms, ignoring case, so "fuzzers" picks the fuzzing entry with all its forms and its style. */ -}} {{- $picked := slice -}} {{- range $page.Param "colorKeywords" -}} {{- $term := . -}} {{- $hit := false -}} {{- range $data.library -}} {{- $entry := . -}} {{- range .match }}{{ if eq (lower .) (lower $term) }}{{ $hit = $entry }}{{ end }}{{ end -}} {{- end -}} {{- if not $hit -}} {{- errorf "%s: color keyword %q isn't in data/keywords.yaml. Add it to the library, or use colorKeywordsCustom = { %q = \"STYLE\" }." $page.File.Path $term $term -}} {{- end -}} {{- $picked = $picked | append (dict "name" $term "match" $hit.match "style" $hit.style) -}} {{- end -}} {{- range $term, $style := $page.Param "colorKeywordsCustom" -}} {{- $picked = $picked | append (dict "name" $term "match" (slice $term) "style" $style) -}} {{- end -}} {{- /* Build one regex per term: any of its forms, whole words only, case-insensitive. Group 1 and 3 are the boundary characters, group 2 is the matched keyword. */ -}} {{- $entries := slice -}} {{- range $i, $k := $picked -}} {{- $style := index $data.styles $k.style -}} {{- if not $style -}} {{- errorf "%s: color keyword %q uses unknown style %q (see styles in data/keywords.yaml)" $page.File.Path $k.name $k.style -}} {{- end -}} {{- $forms := slice -}} {{- range $k.match -}} {{- $escaped := replaceRE `([.+*?()\[\]{}|^$\\])` `\$1` . -}} {{- /* Spaces match any whitespace (Markdown can wrap a phrase across lines), & matches the & it's escaped to in HTML, and an apostrophe matches the curly ’ / ’ that Hugo's typographer turns it into. (& first, since the apostrophe forms contain &.) */ -}} {{- $re := replace (replaceRE `\s+` `\s+` $escaped) "&" "(?:&|&)" -}} {{- $re = replace $re "'" "(?:'|’|’|')" -}} {{- $forms = $forms | append (dict "len" (len .) "re" $re) -}} {{- end -}} {{- $forms = sort $forms "len" "desc" -}} {{- $alternatives := slice -}} {{- range $forms }}{{ $alternatives = $alternatives | append .re }}{{ end -}} {{- $class := printf "kw kw-%s" $k.style -}} {{- if gt (len $style.colors) 1 }}{{ $class = printf "%s kw-grad" $class }}{{ end -}} {{- $entries = $entries | append (dict "i" $i "name" $k.name "len" (index $forms 0).len "class" $class "re" (printf `(?i)(^|[^\p{L}\p{N}_])((?:%s))([^\p{L}\p{N}_]|$)` (delimit $alternatives "|"))) -}} {{- end -}} {{- $entries = sort $entries "len" "desc" -}} {{- /* Walk the tokens. $protTag is the outermost protected element we're inside ("" if none); $depth counts nested elements with that same name so we know when it really closes. */ -}} {{- $s := newScratch -}} {{- $out := "" -}} {{- $protTag := "" -}} {{- $depth := 0 -}} {{- range findRE `|<[^>]*>|[^<]+` $html -}} {{- $tok := . -}} {{- if hasPrefix $tok "<" -}} {{- $isClose := hasPrefix $tok "" -}} {{- $name := lower (replaceRE `^%s` $e.class .) -}} {{- end -}} {{- end -}} {{- return ($out | safeHTML) -}}
