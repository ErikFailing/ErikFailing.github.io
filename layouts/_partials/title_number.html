{{- /* A page's number, shown in front of its title wherever the title is a heading: "#4" on a post, "#1" on a project, "#2" on a series. Called with a page; prints nothing if the page has no `number` in front matter (a series' is in content/series//_index.md). Numbers are set by hand and counted separately for posts, projects and series. The build fails if two pages of the same kind share a number. See "Title numbers" in STYLE.md. */ -}} {{- with .Params.number -}} {{- $number := int . -}} {{- $peers := slice -}} {{- if eq $.Kind "term" }}{{ $peers = $.Parent.Pages }}{{ else }}{{ $peers = where site.RegularPages "Section" $.Section }}{{ end -}} {{- range $peers -}} {{- $peer := . -}} {{- if ne $peer.RelPermalink $.RelPermalink -}} {{- with $peer.Params.number }}{{ if eq (int .) $number -}} {{- errorf "number %d is used by both %s and %s" $number $.RelPermalink $peer.RelPermalink -}} {{- end }}{{ end -}} {{- end -}} {{- end -}} #{{ $number }}{{ " " }} {{- end -}}
