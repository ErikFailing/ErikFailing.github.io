{{- /* Link render hook. Two jobs: 1. Accents: [some phrase](=style) renders the phrase as instead of a link, with STYLE from data/accents.yaml (see "Accents" in STYLE.md). An unknown style fails the build. 2. Every other link renders exactly like Hugo's default Markdown link. */ -}} {{- $dest := .Destination -}} {{- if hasPrefix $dest "=" -}} {{- $name := strings.TrimPrefix "=" $dest -}} {{- $style := index hugo.Data.accents.styles $name -}} {{- if not $style -}} {{- errorf "%s: accent style %q (on %q) isn't in data/accents.yaml" .Page.File.Path $name .PlainText -}} {{- end -}} {{ .Text }} {{- else -}} {{ .Text }} {{- end -}}
