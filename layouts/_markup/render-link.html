{{- /* Link render hook: renders links exactly like Hugo's default Markdown link. Its only extra job is catching the old accent syntax, [text](=style), which would otherwise become a broken link: accents are written {=style: text=} now (see "Accents" in STYLE.md). */ -}} {{- if hasPrefix .Destination "=" -}} {{- errorf "%s: old accent syntax [%s](%s); write it as {=%s: %s=}" .Page.File.Path .PlainText .Destination (strings.TrimPrefix "=" .Destination) .PlainText -}} {{- end -}} {{ .Text }} {{- /* trims the trailing newline */ -}}
